```java
import java.util.Arrays;
import java.util.Locale;
import java.util.Scanner;

public class GradeProject {
    public static void main(String[] args) {
        Locale.setDefault(Locale.US);
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if (n <= 0) {
            System.out.println("暂无成绩");
            return;
        }
        double[] scores = new double[n];
        for (int i = 0; i < scores.length; i++) {
            double score = sc.nextDouble();
            while (score < 0 || score > 100) {
                System.out.println("Invalid input");
                score = sc.nextDouble();
            }
            scores[i] = score;
        }
        double total = 0;
        double highest = scores[0];
        double lowest = scores[0];
        int passed = 0;
        for (double score : scores) {
            total += score;
            if (score > highest) highest = score;
            if (score < lowest) lowest = score;
            if (score >= 60) passed++;
        }
        double[] backup = Arrays.copyOf(scores, scores.length);
        System.out.printf("平均分：%.2f%n", total / n);
        System.out.printf("最高分：%.2f%n", highest);
        System.out.printf("最低分：%.2f%n", lowest);
        System.out.println("及格人数：" + passed + "/" + n);
        // 周日实验：临时加入 scores[0] = 0; 再观察 backup[0]。
        // 不应影响 backup；实验后移除额外输出再跑自测。
    }
}
```

- 人数不大于 0 时直接结束，避免空数组访问和除以 0。
- 录入用普通 for，统计可用 for-each。
- Arrays.copyOf 对本例一维数值数组建立独立副本。
