```java
import java.util.Locale;
import java.util.Scanner;

public class CourseReport {
    public static void main(String[] args) {
        Locale.setDefault(Locale.US);
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        if (n <= 0 || m <= 0) {
            System.out.println("暂无成绩");
            return;
        }
        double[][] scores = new double[n][m];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                scores[i][j] = sc.nextDouble();
        for (int j = 0; j < m; j++) {
            double total = 0;
            for (int i = 0; i < n; i++) total += scores[i][j];
            System.out.printf("课程%d平均分：%.2f%n", j + 1, total / n);
        }
    }
}
```

- 学生行、课程列的含义保持一致。
- 按课程统计时，外层课程、内层学生；每科重新开始累加。
